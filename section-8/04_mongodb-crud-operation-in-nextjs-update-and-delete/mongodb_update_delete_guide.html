Dynamic Mongoose Operations: app/todos/[id]/route.js

Querying, updating, and removing documents by their MongoDB ObjectId

app/todos/[id]/route.js GET, PUT & DELETE Methods 
import Todo from "@/models/todoModel";
import { connectDB } from "@/lib/connectDB";

// 🟢 GET SINGLE TODO
export async function GET(_, { params }) {
  await connectDB();
  const { id } = await params;

  const todo = await Todo.findById(id);
  if (!todo) {
    return Response.json({ error: "Todo not found" }, { status: 404 });
  }
  return Response.json(todo);
}

// 🔄 UPDATE TODO (PUT)
export async function PUT(request, { params }) {
  await connectDB();
  const editTodoData = await request.json();
  const { id } = await params;

  // ⚠️ CRUCIAL: { new: true } returns updated document!
  const editedTodo = await Todo.findByIdAndUpdate(id, editTodoData, {
    new: true,
  });

  return Response.json(editedTodo);
}

// 🗑️ DELETE TODO
export async function DELETE(_, { params }) {
  await connectDB();
  const { id } = await params;

  await Todo.findByIdAndDelete(id);

  // 🔴 RFC 204 No Content with strictly null body
  return new Response(null, {
    status: 204,
  });
}
Why is { new: true } So Crucial in Mongoose?

By default, Mongoose ka findByIdAndUpdate() method database mein update karne ke baad OLD (pre-update) document return karta hai!
Agar aap { new: true } nahi lagayenge, toh backend API client ko purani value bhej degi, jisse frontend UI confuse ho jayegi. { new: true } pass karne se Mongoose freshly updated document return karta hai.
